    def rk(v):
        o=sorted(range(len(v)),key=lambda i:v[i]);r=[0]*len(v)
        for i,x in enumerate(o):r[x]=i
        return r
    ra,rb=rk(a),rk(b);n=len(a);ma=(n-1)/2
    return sum((x-ma)*(y-ma) for x,y in zip(ra,rb))/math.sqrt(sum((x-ma)**2 for x in ra)*sum((y-ma)**2 for y in rb))
