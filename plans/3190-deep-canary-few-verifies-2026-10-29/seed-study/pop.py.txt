import os,subprocess,time,collections
W='/home/leo/src/warm-lanes/worktrees'
c=[]
for n in os.listdir(W):
    if n.startswith(('_mainprobe-','_mainsweep-')):
        p=f'{W}/{n}/target'
        if os.path.isdir(p):
            st=os.stat(f'{W}/{n}');c.append((st.st_mtime,n))
c.sort()
step=max(1,len(c)//16);pick=c[::step]+c[-1:]
extra=[('/home/leo/src/warm-lanes/personal/gui-dev-20261008','gui-dev-20261008'),('/home/leo/src/warm-lanes/personal/printer-dogfood-20260823','printer-dogfood-20260823')]
print('tree | birth(stat %W UTC) | entries-older-than-birth: total f d fp_entries fp_units dwo debug_deps release_entries | later-entries(own builds)')
def scan(path,name):
    b=int(subprocess.run(['stat','-c','%W',path],capture_output=True,text=True).stdout or 0)
    if b<=0:b=int(os.stat(path).st_mtime)
    t=f'{path}/target'
    p=subprocess.Popen(['find',t,'-xdev','-printf','%y %T@ %P\n'],stdout=subprocess.PIPE,text=True,errors='replace')
    tot=f=d=fp=fu=dwo=dd=rel=later=0
    for l in p.stdout:
        y,m,q=l.rstrip('\n').split(' ',2)
        if float(m)>=b:
            later+=1;continue
        tot+=1;f+=y=='f';d+=y=='d'
        a=q.split('/')
        if len(a)>=3 and a[1]=='.fingerprint':
            fp+=1
            if len(a)==3 and y=='d':fu+=1
        if q.endswith('.dwo'):dwo+=1
        if len(a)>=3 and a[0]=='debug' and a[1]=='deps':dd+=1
        if a[0]=='release':rel+=1
    print(name,time.strftime('%m-%d %H:%M',time.gmtime(b)),tot,f,d,fp,fu,dwo,dd,rel,'|',later,flush=True)
for m,n in pick:scan(f'{W}/{n}',n)
for p,n in extra:
    if os.path.isdir(p+'/target'):scan(p,n)
