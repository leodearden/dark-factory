import json,re,time,calendar,collections,math
def pct(xs,q):
    xs=sorted(xs);k=q*(len(xs)-1);f=int(k);c=min(f+1,len(xs)-1);return xs[f]+(xs[c]-xs[f])*(k-f)
# sar timestamps are host local: BST(+1) through 10-24, GMT from 10-25 02:00 local
def load(fn):
    O={}
    for l in open(fn):
        d,h,ap,tps,r,w,dk,aq,aw,ut=l.split()
        H,M,S=map(int,h.split(':'))
        H=(H%12)+(12 if ap=='PM' else 0)
        t=calendar.timegm((2026,10,int(d),H,M,S))
        if int(d)<25 or (int(d)==25 and H<2): t-=3600
        O[t]=dict(tps=float(tps),r=float(r),w=float(w),d=float(dk),aq=float(aq),aw=float(aw),ut=float(ut))
    return O
L=load('/tmp/3190s/sar_loop0.txt');D=load('/tmp/3190s/sar_dm0.txt')
S=json.load(open('/tmp/3190s/phases.json'))['seeds']
b=collections.defaultdict(list);rl=collections.defaultdict(list)
for s in S:
    if 'clone0' in s and 'clone1' in s:
        # sar stamp is END of its 600s interval; assign by clone midpoint
        m=(s['clone0']+s['clone1'])/2;k=int(m//600+1)*600
        b[k].append(s['clone1']-s['clone0'])
        if 'relink' in s and 'reloc' in s: rl[k].append(s['relink']-s['reloc'])
X=[(k,L[k],D.get(k),pct(b[k],.5),len(b[k]),pct(rl[k],.5) if rl[k] else None) for k in sorted(b) if k in L]
print('10-min bins with >=1 clone',len(X))
def spear(a,b):
    def rk(v):
        o=sorted(range(len(v)),key=lambda i:v[i]);r=[0]*len(v)
        for i,x in enumerate(o):r[x]=i
        return r
    ra,rb=rk(a),rk(b);n=len(a);ma=(n-1)/2
    return sum((x-ma)*(y-ma) for x,y in zip(ra,rb))/math.sqrt(sum((x-ma)**2 for x in ra)*sum((y-ma)**2 for y in rb))
for f in ('tps','r','w','aq','aw','ut'):
    print('spearman clone_p50 ~ loop0',f,round(spear([x[3] for x in X],[x[1][f] for x in X]),2),' dm0',f,round(spear([x[3] for x in X if x[2]],[x[2][f] for x in X if x[2]]),2))
