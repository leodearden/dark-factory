import json,datetime as dt,collections,statistics as st,re
U=dt.timezone.utc
def T(s): return dt.datetime.fromisoformat(s).astimezone(U)
A=json.load(open('a1.json'))
free=[(T(t),f) for t,f in A['free']]
# gross flows
dn=sum(max(0,free[i][1]-free[i+1][1]) for i in range(len(free)-1));up=sum(max(0,free[i+1][1]-free[i][1]) for i in range(len(free)-1))
days=(free[-1][0]-free[0][0]).total_seconds()/86400
print('days',round(days,2),'gross decline GiB/day',round(dn/days),'gross rise GiB/day',round(up/days),'net',free[-1][1]-free[0][1])
by=collections.defaultdict(list)
for t,f in free: by[t.date()].append(f)
for d,v in sorted(by.items()): print(d,'min',min(v),'max',max(v),'p50',int(st.median(v)))
# worst decline over windows
for H in (1,6,24):
    w=dt.timedelta(hours=H);best=(0,None);j=0
    for i in range(len(free)):
        while free[j][0]<free[i][0]-w: j+=1
        dd=max(f for _,f in free[j:i+1])-free[i][1]
        if dd>best[0]: best=(dd,free[i][0])
    print('worst decline in',H,'h:',best[0],'GiB at',best[1])
# delta free per timer pass
s=A['sums'];import bisect
ts=[t for t,_ in free]
def at(t):
    i=bisect.bisect_left(ts,t);i=min(i,len(ts)-1);return free[i][1]
row=[]
for i in range(1,len(s)):
    a=T(s[i-1][0]);b=T(s[i][0]);row.append((a.strftime('%m-%d %H:%M'),round((b-a).total_seconds()/3600,1),at(a),at(b),at(b)-at(a),s[i][2]))
print('per-pass Δfree median',st.median(r[4] for r in row),'min',min(r[4] for r in row),'max',max(r[4] for r in row))
for r in row[-6:]: print(r)
