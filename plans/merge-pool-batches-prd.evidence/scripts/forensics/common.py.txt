def canon(t):
    m=re.search(r'/(test_[A-Za-z0-9_]+)\.py',t)
    b=m.group(1) if m else t
    return CAN.get(b,b.replace('test_','')[:22])
def fset(v):
    o=F.get(str(v['vid']))
    if o is None or not o['files']: return None
    s=set(canon(t) for t in o['failed'])
    if not s and o['cats']: s=set('<'+c+'>' for c in o['cats'])
    if o.get('timed_out'): s.add('<timeout>')
    return s
