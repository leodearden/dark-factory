import sqlite3, json
c=sqlite3.connect('file:/home/leo/src/dark-factory/data/orchestrator/runs.db?mode=ro',uri=True,timeout=60)
def row(i):
    r=c.execute("select id,timestamp,task_id,event_type,data from events where id=?",(i,)).fetchone()
    d=json.loads(r[4]); u=d.pop('units',None)
    if u is not None: d['RED_UNITS']=[x['unit'] for x in u if x['outcome']!='ran_green']; d['n_units']=len(u)
    for k in ('plan_hash','cache_mode','chain_build_ms','retry_scope','retry_subset_sizes','runner_reason','waiting','overlap_files','snapshot_tip','superseded_by','generation','absorbed_request_ids','attempt','chain_depth_reason'): d.pop(k,None)
    if 'reason' in d and d['reason']: d['reason']=d['reason'][:90]
    print('  ',r[0],r[1][:19],r[2],r[3],json.dumps(d)[:470])
def win(task,lo,hi,types=('merge_verify','merge_finalized','merge_attempt','speculative_merge','speculative_discard','verdict_voided','merge_verify_cancelled')):
    q="select id from events where task_id=? and timestamp>=? and timestamp<=? and event_type in (%s) order by id"%','.join('?'*len(types))
    for (i,) in c.execute(q,(task,lo,hi)+tuple(types)): row(i)
