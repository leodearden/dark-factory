def ts(s):
    return datetime.fromisoformat(s).timestamp()

def by_type(t, lo='2026-10-20'):

GATE_SHAS = defaultdict(set)
for g in by_type('merge_reverify_gate'):
    if g['d'].get('merge_sha'):

def verifies(lo=T0):
    out = []
    for e in by_type('merge_verify', lo):
        d = e['d']
        if d.get('runner') == 'history-leg':
            continue
        k = d.get('chain_items') or 1
        out.append({
            'id': e['id'], 'ts': e['ts'], 'task': e['task'], 'k': k,
            'passed': d.get('passed') is True, 'runner': d.get('runner'),
            'dur_s': (d.get('duration_ms') or 0) / 1000.0,
            'start_ts': ts(e['ts']) - (d.get('duration_ms') or 0) / 1000.0,
            'end_ts': ts(e['ts']),
            'gate': d.get('merge_sha') in GATE_SHAS,
            'gate_same_task': e['task'] in GATE_SHAS.get(d.get('merge_sha'), ()),
            'runner_reason': d.get('runner_reason'),
            'members': d.get('chain_task_ids'),
            'req_ids': d.get('chain_request_ids'),
            'cdr': d.get('chain_depth_reason', 'ABSENT'),
            'merge_sha': d.get('merge_sha'),
            'win': window(e['ts']), 'day': e['ts'][:10],
            'n_red': d.get('_n_red'), 'red_units': d.get('_red_units'),
        })
    return out


def pct(vals, p):
    if not vals:
        return None
    s = sorted(vals)
    if len(s) == 1:
        return s[0]
    k = (len(s) - 1) * p / 100
    f, c = math.floor(k), math.ceil(k)
    return s[f] + (k - f) * (s[c] - s[f])


def wilson(x, n, z=1.96):
    if n == 0:
        return (None, None)
    p = x / n
    den = 1 + z * z / n
    c = (p + z * z / (2 * n)) / den
    h = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / den
    return (c - h, c + h)


def binom_cdf(x, n, p):
    return sum(math.comb(n, i) * p ** i * (1 - p) ** (n - i) for i in range(x + 1))


def cp_upper(x, n, alpha=0.05):
    """One-sided 95% Clopper-Pearson upper bound on p given x of n."""
    if x >= n:
        return 1.0
    lo, hi = x / n, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if binom_cdf(x, n, mid) > alpha:
            lo = mid
        else:
            hi = mid
    return hi


def fmt(x, d=3):
    return '-' if x is None else (f'{x:.{d}f}' if isinstance(x, float) else str(x))
