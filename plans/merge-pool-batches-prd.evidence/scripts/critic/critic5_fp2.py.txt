def trace(q,r,t,u,mf=12,mt=24,scan='auto',n=60):
    a_f,a_c=1.0,1/mt; h=[]
    for i in range(n):
        na_c=min(cost(0,c,q,r,t,u,a_f,a_c,fb_release,scan)/c for c in range(1,mt+1))
        na_f=min(cost(f,0,q,r,t,u,a_f,na_c,fb_release,scan)/f for f in range(1,mf+1))
        h.append((round(na_f,4),round(na_c,4)))
        if abs(na_c-a_c)<1e-9 and abs(na_f-a_f)<1e-9: return h,True
        a_f,a_c=na_f,na_c
    return h,False
