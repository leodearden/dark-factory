import sqlite3, json, collections
REPO='/home/leo/src/dark-factory'
db=sqlite3.connect(f'file:{REPO}/data/orchestrator/runs.db?mode=ro', uri=True)
tips={}
for id_,ts,task,data in db.execute("""select id,timestamp,task_id,data from events where event_type='merge_verify'
  and timestamp>='2026-10-21' and json_extract(data,'$.chain_items')>1 and json_extract(data,'$.passed')=0"""):
    d=json.loads(data); tips[d['merge_sha']]=(ts,d.get('runner'),d.get('chain_items'),d.get('units'),d.get('duration_ms'))
per=collections.defaultdict(set); ids=collections.defaultdict(list)
for test_id,verdict,sha,detail in db.execute("select test_id,verdict,merge_sha,detail from flake_occurrence where call_site='merge_gate' and observed_at>='2026-10-21'"):
    if sha in tips:
        try: r=json.loads(detail).get('unconfirmable_reason')
        except Exception: r=None
        per[sha].add((verdict,r)); ids[sha].append(test_id)
c=collections.Counter()
for sha,s in per.items():
    if any(v=='fails_in_isolation' for v,_ in s): continue
    c[tuple(sorted(str(r) for _,r in s))]+=1
    if any(r=='node_ids_unmapped_to_subproject' for _,r in s): print('unmapped ids sample', ids[sha][:4])
print('unconfirmable tips by reason:',c)
# units: red legs per red tip
uc=collections.Counter(); mixed=0; legs_only=0; n_units=0
for sha,(ts,runner,n,units,dur) in tips.items():
    if not units: uc['no units']+=1; continue
    n_units+=1
    red=[u for u in units if isinstance(u,dict) and u.get('outcome')=='ran_red']
    kinds=tuple(sorted({u['unit'].split(':')[-1] for u in red}))
    uc[kinds]+=1
print('red legs by kind (tips with units=%d):'%n_units, uc)
print('unit sample', [u for u in next(v[3] for v in tips.values() if v[3])][:3])
# runner split + scan-relevant: how many red tips ran on laptop
print(collections.Counter(v[1] for v in tips.values()))
