"""Checks of draft 2's rule (6.5) against the simulated policy's own functions (read-only import)."""
import sys
sys.path.insert(0, '/tmp/gate-10069/sim')
import lane as L


def depth_U(p, e, A, follow_one, kmax=12):
    a = lambda x: A[min(len(A) - 1, int(round(x * (len(A) - 1))))]
    def C(k):
        R = 1 - (1 - p) ** k * (1 - e)
        if k == 1:
            return 1.0
        return 1 + R * k * (1.0 if follow_one else a(min(1.0, p / R)))
    k = 1
    while k < kmax and C(k + 1) - C(k) < a(p):
        k += 1
    return k

    row = [(L.table_row(est, L.TABLE_NEW), depth_U(1 - est, e, A, True)) for est in ests]
    bad = [(est, t, u) for est, (t, u) in zip(ests, row) if t != u]
