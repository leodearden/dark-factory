from rule7 import fallback_cost, batch_cost, going_costs, plan_batch
R,U=0.01,0.15
print('cleared front, k cleared waiting in all (incl. front), 20 new waiting; starting rates; scanning on')
print(' q    k   plan     C(plan)  best alt (f,c)  saving@going  marginal 1st new vs A_new')
for q in (0.05,0.08,0.11,0.14,0.17,0.20,0.30):
    a_new,a_cl,_=going_costs(q,R,U,True)
    C=lambda f,c: batch_cost(f,c,q,R,U,True,a_new,a_cl)
    for k in (1,2,3,6,11):
        f,c=plan_batch('cleared',['cleared']*(k-1)+['new']*20,q,R,U,True)
        best=0;arg=(f,c)
        for j in range(f+1,min(12,24-c)+1):
            s=(j-f)*a_new-(C(j,c)-C(f,c))
            if s>best: best,arg=s,(j,c)
        print(f'{q:.2f} {k:3d}  {(f,c)!s:8} {C(f,c):.3f}   {arg!s:8}       {best:.3f}        {C(f+1,c)-C(f,c):.3f} vs {a_new:.3f}')
