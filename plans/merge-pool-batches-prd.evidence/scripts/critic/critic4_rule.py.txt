import math
TABLE=[(0.992,12),(0.990,11),(0.988,10),(0.984,9),(0.979,8),(0.972,7),(0.959,6),(0.934,5),(0.876,4),(0.75,3)]
def table_depth(p):
    for thr,d in TABLE:
        if p>=thr: return d
    return 1
def P_red(f,c,q,r): return 1-(1-q)**f*(1-r)**c
def C(f,c,q,r,t,u,a0):
    n=f+c
    if n<=1: return 1.0
    pr=P_red(f,c,q,r)
    if pr<=0: return 1.0
    sound=n-(f*q+c*r)/pr
    return 1+pr*((1-u)*(t*(n+1)+sound*a0)+u*n)
def going(q,r,t,u,maxf):
    a0=1/maxf
    return min(C(k,0,q,r,t,u,a0)/k for k in range(1,maxf+1)), a0
def plan(nfresh_avail,ncleared_avail,q,r,t,u,maxf,maxt,order='fresh_first'):
    aq,a0=going(q,r,t,u,maxf)
    f,c=1,0   # dispatching request is first member (fresh)
    while f<maxf and f-1<nfresh_avail-1:
        if C(f+1,c,q,r,t,u,a0)-C(f,c,q,r,t,u,a0) < aq: f+=1
        else: break
    took=0
    while f+c<maxt and took<ncleared_avail:
        if C(f,c+1,q,r,t,u,a0)-C(f,c,q,r,t,u,a0) < a0: c+=1; took+=1
        else: break
    return f,c,aq,a0
print('== B1: u=1 vs depth table (max_fresh 12), pass estimate p=1-q')
mism=[]
p=0.75
while p<=0.9951:
    q=1-p
    f,c,aq,a0=plan(100,0,q,0.01,0.05,1.0,12,24)
    td=table_depth(p)
    if f!=td: mism.append((round(p,4),td,f))
    p+=0.0005
print('grid points',int((0.995-0.75)/0.0005)+1,'mismatches',len(mism))
# summarise mismatch ranges
if mism:
    runs=[];s=mism[0];prev=mism[0]
    for m in mism[1:]:
        if m[1:]!=prev[1:] or m[0]-prev[0]>0.00051: runs.append((s[0],prev[0],prev[1],prev[2])); s=m
        prev=m
    runs.append((s[0],prev[0],prev[1],prev[2]))
    for r_ in runs: print('  p in [%.4f,%.4f]: table %d, rule %d'%r_)
print('== rule depth at u=1 for sample p:', [(p,plan(100,0,1-p,0.01,0.05,1.0,12,24)[0],table_depth(p)) for p in (0.75,0.80,0.86,0.876,0.90,0.934,0.95,0.959,0.972,0.979,0.984,0.988,0.99,0.992,0.995)])

print('== u=0 (sim cost) ')
