"""L15 numbers for draft 3 (section 6.6)."""
import random
from math import comb

B = lambda n, k, p: comb(n, k) * p ** k * (1 - p) ** (n - k)
atleast2 = lambda p: 1 - B(20, 0, p) - B(20, 1, p)

print('== draft: "catches a 30% wrong rate 82% of the time and trips at a 3% wrong rate 4% of the time" (half return unchanged)')
print('  w=0.30 r=0.5: %.3f ; w=0.03 r=0.5: %.3f' % (atleast2(0.15), atleast2(0.015)))
print('== at the wrong share the simulation itself reports (0.28 wrong : ~4.5 right per 100 at 0.14 -> 5.9%; 0.094 by construction at 0.08)')
for w in (0.059, 0.094):
    for r in (1.0, 0.5):
        print('  w=%.3f r=%.1f: P(2 or more of 20) = %.3f' % (w, r, atleast2(w * r)))

random.seed(7)


def run(w, r, horizon=2000):
    """Convictions until OFF: ON -> SUSPENDED at 2 sound truths among the newest 20 on-convictions;
    re-qualify (assumed to succeed); second suspension -> OFF. Returns (first suspension, off) counts."""
    first = None
    win = []
    susp = 0
    for n in range(1, horizon + 1):
        win.append(random.random() < w * r)
        win = win[-20:]
        if sum(win) >= 2:
            susp += 1
            if susp == 1:
                first = n
                win = []
            else:
                return first, n
    return first, None


print('== sliding window of the newest 20: convictions until the first suspension, and until OFF (median over 4000 runs)')
for w, r in ((0.059, 1.0), (0.059, 0.5), (0.094, 1.0), (0.094, 0.5), (0.03, 0.5)):
    fs, offs = [], []
    for _ in range(4000):
        f, o = run(w, r)
        fs.append(f if f is not None else 10 ** 9)
        offs.append(o if o is not None else 10 ** 9)
    fs.sort(); offs.sort()
    med = lambda v: v[len(v) // 2]
    print('  w=%.3f r=%.1f: first suspension median %s ; OFF median %s ; OFF within 300 convictions %.2f'
          % (w, r, med(fs), med(offs), sum(1 for o in offs if o <= 300) / len(offs)))

print('== SHADOW -> ON, cumulative since the last state row: n >= 10 and sound <= n/10, checked at every truth')
for prec in (0.95, 0.906, 0.85, 0.8, 0.7, 0.6):
    for N in (10, 30, 100):
        hit = 0
        for _ in range(20000):
            s = 0
            for n in range(1, N + 1):
                s += random.random() > prec
                if n >= 10 and s * 10 <= n:
                    hit += 1
                    break
        print('  precision %.3f, within %3d truths: %.2f' % (prec, N, hit / 20000), end=';')
    print()
