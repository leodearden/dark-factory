def qhat(q,f,u=U):
    # per exposure of a new request in a batch of f new: convicted q (scannable share), landed-as-new (1-q)^f
    # unscannable red (share u of reds): fallback alone -> blocked q / landed-as-new: sound ones ahead of first culprit; rest re-exposed
    pass_ = (1-q)**f
    ev = q*(1-u) + 0   # convictions
    land_new = pass_
    # fallback part: members verified alone in order until first blocked: blocked=1 event; B sound landed as new
    Pred=1-pass_
    p=q; rn=Pred
    B=sum((j-1)*(1-p)**(j-1)*p for j in range(1,f+1))/rn
    ev_fb = Pred*u*1.0/f ; land_fb = Pred*u*B/f
    return (ev+ev_fb)/(ev+ev_fb+land_new+land_fb)
