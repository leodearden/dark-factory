"""Exact enumeration of the draft's 6.3 model, for hand-constructed cases.

observation = (frozenset(keys), 'red'|'pass').  A pass clears its keys.
"""
from itertools import product


def infer(obs, d, f, voided=()):
    cleared = set()
    for i, (keys, v) in enumerate(obs):
        if v == 'pass':
            cleared |= set(keys)
    reds = [set(k) - cleared for i, (k, v) in enumerate(obs) if v == 'red' and i not in voided]
    unexplained = sum(1 for r in reds if not r)
    reds = [r for r in reds if r]
    keys = sorted(set().union(*reds)) if reds else []
    tot = 0.0
    marg = {k: 0.0 for k in keys}
    for bits in product([0, 1], repeat=len(keys)):
        bad = {k for k, b in zip(keys, bits) if b}
        p = 1.0
        for k, b in zip(keys, bits):
            p *= d if b else (1 - d)
        for r in reds:
            p *= 1.0 if (r & bad) else f
        tot += p
        for k in bad:
            marg[k] += p
    return {k: (marg[k] / tot if tot else float('nan')) for k in keys}, unexplained


def show(title, obs, d=0.14, f=0.01, voided=()):
    post, unx = infer(obs, d, f, voided)
    print(f'{title}  [d={d} f={f}] unexplained={unx}')
    for k, v in post.items():
        flag = 'SUSPECT' if v > 1.2 * d else ''
        conv = 'CONVICT' if v >= 0.9 else ''
        print(f'   {k}: {v:.4f} {flag} {conv}')

    # threshold check by batch size
    for n in (2, 3, 4, 5, 6, 8, 12):
        post, _ = infer([([f'k{i}' for i in range(n)], 'red')], 0.14, 0.01)
        print(f'batch {n} red on main: each {list(post.values())[0]:.3f}  suspect thr {1.2*0.14:.3f}')
