import sqlite3, json, re, sys
con = sqlite3.connect("file:/home/leo/src/dark-factory/.taskmaster/tasks/tasks.db?mode=ro", uri=True)
hits = json.load(open("hits.json"))
want = set(int(a) for a in sys.argv[1:])
for i, s, t, terms, indraft in hits:
    if i not in want: continue
    title, desc, det = con.execute("select title, description, details from tasks where id=?", (i,)).fetchone()
    text = (title or "") + " || " + (desc or "") + " || " + (det or "")
    text = text.replace("\n", " ")
    print(f"## {i}")
    for term in terms:
        ms = [m.start() for m in re.finditer(re.escape(term), text, flags=re.I)]
        for p in ms[:2]:
            print(f"  [{term}] ...{text[max(0,p-230):p+260]}...")
    print()
