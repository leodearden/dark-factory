R,U=0.01,0.15
# 1. the lone new member of a mixed batch: F(1)=0 in the formula
def fb1(n,p,a_new):
    return 1.0 if n==1 else fallback_cost(n,p,a_new)
print('cleared front, 11 cleared, 20 new: plan with F(1)=0 (contract) vs F(1)=1')
for q in (0.08,0.11,0.14,0.20):
    a,c,_=going_costs(q,R,U,True)
    m0=batch_cost(1,11,q,R,U,True,a,c)-batch_cost(0,11,q,R,U,True,a,c)
    m1=batch_cost(1,11,q,R,U,True,a,c,fallback=fb1)-batch_cost(0,11,q,R,U,True,a,c,fallback=fb1)
    print(f'q={q}: marginal first new {m0:.3f} (F(1)=0) / {m1:.3f} (F(1)=1) vs A_new {a:.3f}; plans',
          plan_batch('cleared',['cleared']*10+['new']*20,q,R,U,True), plan_batch('cleared',['cleared']*10+['new']*20,q,R,U,True,fallback=fb1))
# 2. convergence of the draft-8 formula at corners, caps, starts
def gc_from(q,r,u,sc,mf,mt,start):
    a_new,a_cl=start
    for i in range(500):
        nc=min(batch_cost(0,c,q,r,u,sc,a_new,a_cl)/c for c in range(1,mt+1))
        nf=min(batch_cost(f,0,q,r,u,sc,a_new,nc)/f for f in range(1,mf+1))
        if abs(nc-a_cl)<1e-9 and abs(nf-a_new)<1e-9: return round(nf,6),round(nc,6),i
        a_new,a_cl=nf,nc
    return None
bad=multi=cells=mx=0
for q in (1e-9,1e-4,0.01,0.05,0.14,0.3,0.5,0.7,0.9,0.99,0.999999):
  for r in (0.0,1e-6,0.01,0.1,0.5,0.9):

      for sc in (True,False):
        for mf,mt in ((12,24),(1,1),(1,24),(4,4),(24,48),(48,96),(12,12),(2,3)):
          cells+=1
          res=[gc_from(q,r,u,sc,mf,mt,s) for s in ((1.0,1.0),(0.01,0.01),(5.0,5.0),(1.0,0.04))]
          if any(x is None for x in res): bad+=1; print('NOCONV',q,r,u,sc,mf,mt) if bad<6 else None; continue
          mx=max(mx,max(x[2] for x in res))
          if len({x[:2] for x in res})>1:
              multi+=1; print('MULTI',q,r,u,sc,mf,mt,res) if multi<6 else None
print('cells',cells,'nonconv',bad,'start-dependent',multi,'max iters',mx)
