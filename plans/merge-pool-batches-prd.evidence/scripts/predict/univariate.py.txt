import json,collections,datetime as dt,sys
P='/tmp/gate-10069/predict/'
rows=json.load(open(P+'rows.json'))
def T(s): return dt.datetime.fromisoformat(s).timestamp()
LO=T('2026-10-22T00:00:00+00:00'); HI=T('2026-11-20T00:00:00+00:00')
D8528=T('2026-11-19T04:15:00+00:00')
rows=[r for r in rows if LO<=r['verdict_t']<HI]
def era(r):
    t=r['verdict_t']
    if t<T('2026-11-01T08:00:00+00:00'): return 'E1 10-22..11-01 trains'
    if t<T('2026-11-11T00:00:00+00:00'): return 'E2 11-01..11-11 singles'
    if t<D8528: return 'E3 11-11..11-19 chains'
    return 'E4 post-8528'
for r in rows: r['era']=era(r)
print('WINDOW 10-22..11-19; rows',len(rows),'fail',sum(r['y'] for r in rows),'rate %.3f'%(sum(r['y'] for r in rows)/len(rows)))
print('by src',collections.Counter(r['src'] for r in rows))
def tab(name,f,pop=rows,order=None):
    g=collections.defaultdict(lambda:[0,0])
    for r in pop:
        k=f(r); g[k][0]+=1; g[k][1]+=r['y']
    keys=order or sorted(g,key=lambda k:(k is None,str(k)))
    print(f'-- {name}')
    for k in keys:
        if k in g:
            n,y=g[k]; print(f'   {str(k):28s} n={n:5d} fail={y:4d} rate={y/n:.3f}')
def b(v,edges):
    if v is None: return None
    for i,e in enumerate(edges):
        if v<e: return f'{i}:<{e}'
    return f'{len(edges)}:>={edges[-1]}'
tab('era',lambda r:r['era'])
tab('era x src',lambda r:(r['era'][:2],r['src']))
solo=[r for r in rows if r['src'] in('solo_pass','solo_fail')]
for label,pop in (('ALL labelled requests',rows),('SOLO-verdict requests only',solo)):
    print('\n==========',label,'n=',len(pop),'fail rate %.3f'%(sum(r['y'] for r in pop)/len(pop)))
    tab('prior_fail (earlier requests of same task with solo fail)',lambda r:min(r['prior_fail'],3),pop)
    tab('resub',lambda r:r['resub'],pop)
    tab('prior_reqs',lambda r:min(r['prior_reqs'],4),pop)
    tab('behind_enq (commits base..main at enqueue)',lambda r:b(r['behind_enq'],[1,5,15,40,100]),pop)
    tab('behind_start (commits base..main at verify start)',lambda r:b(r['behind_start'],[1,5,15,40,100]),pop)
    tab('queue_wait_h',lambda r:b(r['queue_wait_h'],[0.25,1,3,8,24]),pop)
    tab('task_verify_age_h (last task verify -> merge verify start)',lambda r:b(r['task_verify_age_h'],[0.5,2,6,24]),pop)
    tab('n_task_files',lambda r:b(r['n_task_files'],[3,6,12,25]),pop)
    tab('n_modules',lambda r:b(r['n_modules'],[2,3,5]),pop)
    tab('workspace scope',lambda r:r['workspace'],pop)
    tab('wv_units',lambda r:b(r['wv_units'],[4,7,13]),pop)
    tab('n_task_verifies',lambda r:b(r['n_task_verifies'],[1,2,3,5]),pop)
    tab('n_rebases',lambda r:b(r['n_rebases'],[1,2,4]),pop)
    tab('rb_cohort',lambda r:r['rb_cohort'],pop)
    tab('rb_distance',lambda r:b(r['rb_distance'],[5,15,40,100]),pop)
    tab('priority (task store)',lambda r:r['priority'],pop)
    tab('lane (>=11-15)',lambda r:r['lane'],pop)
    tab('tier (>=11-15)',lambda r:r['tier'],pop)
    tab('task_kind',lambda r:r['task_kind'],pop)
    tab('complexity',lambda r:r['complexity'],pop)
    tab('spawn_context',lambda r:r['spawn_context'],pop)
    tab('human_decomposed',lambda r:r['human_decomposed'],pop)
    tab('has_prd',lambda r:r['has_prd'],pop)
    tab('n_decl_files',lambda r:b(r['n_decl_files'],[1,3,6,12]),pop)
    tab('numeric_branch',lambda r:r['numeric_branch'],pop)
    tab('hour UTC (4h)',lambda r:'%02d-%02d'%(r['hour']//4*4,r['hour']//4*4+4),pop)
    tab('runner',lambda r:r['runner'],pop)
    tab('base_relation (>=11-07)',lambda r:r['base_relation'],pop)
    tab('speculative',lambda r:r['speculative'],pop)
    tab('conc (verifies in flight at start)',lambda r:b(r['conc'],[1,2,3,5]),pop)
    tab('prev_red_batch',lambda r:r['prev_red_batch'],pop)
    tab('red_chain_before',lambda r:min(r['red_chain_before'],3),pop)
    tab('lane_fail30 (rolling solo fail rate, last 30 before start)',lambda r:b(r['lane_fail30'],[0.05,0.15,0.3,0.5]),pop)
    tab('table_est (8528-style per-item pass estimate)',lambda r:b(r['table_est'],[0.6,0.75,0.85,0.93]),pop)
json.dump(rows,open(P+'rows_window.json','w'))
