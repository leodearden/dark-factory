import json,collections
E=[json.loads(l) for l in open('/tmp/gate-10069/predict/events.jsonl')]
mv=[e for e in E if e['et']=='merge_verify']
keys=collections.Counter()
first={}
for e in mv:
    for k,v in e['d'].items():
        if v is not None:
            keys[k]+=1; first.setdefault(k,e['ts'])
for k in keys: print(k,keys[k],first[k])
print()
byday=collections.defaultdict(lambda: collections.Counter())
for e in mv:
    d=e['d']; day=e['ts'][:10]
    k=d.get('chain_items') or 1
    byday[day][('k1' if k==1 else 'k>1', bool(d.get('passed')))]+=1
for day in sorted(byday):
    c=byday[day]; print(day, 'k1 pass/fail',c[('k1',True)],c[('k1',False)],'  chain pass/fail',c[('k>1',True)],c[('k>1',False)])
print(collections.Counter(e['d'].get('runner') for e in mv))
print(collections.Counter(e['d'].get('base_relation') for e in mv))
print(collections.Counter(e['d'].get('runner_reason') for e in mv))
print(collections.Counter(str(e['d'].get('retry_scope')) for e in mv))
print(collections.Counter(e['d'].get('attempt') for e in mv))
print(collections.Counter(e['d'].get('depth') for e in mv))
