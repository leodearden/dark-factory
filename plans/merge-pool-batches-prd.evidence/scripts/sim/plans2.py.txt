QS = (0.05, 0.08, 0.14, 0.20)
TD = {0.02: 8, 0.03: 6, 0.05: 5, 0.08: 4, 0.14: 3, 0.20: 3}
DEPTHS = [('+0', {}), ('+1', dict(cap_add=1)), ('+2', dict(cap_add=2)), ('+3', dict(cap_add=3)), ('+4', dict(cap_add=4)),
          ('+6', dict(cap_add=6)), ('x2', dict(cap_mult=2))]
ISO = [('J1d', dict(sus='J1d')), ('J2', dict(sus='J2', tau=0.5)), ("J2'", dict(sus='J2', tau=0.65))]
NODL = dict(deadline_mult=99)


def J(**kw):
    d = dict(base='J', e_pol=0.009, theta=0.9)
    d.update(kw)
    return d


def cell(q, dur='W1', env='on', tag='', mode='cap', **extra):
    cfg = dict(q=q, e=0.01, dur=dur, horizon=144.0, warm=24.0)
    if mode == 'cap':
        cfg.update(closed_N=75)
    if env == 'on':
        cfg.update(mainred=True, restarts=True)
    cfg.update(extra)
    return '%s|q=%.2f|%s|env=%s%s' % (mode, q, dur, env, ('|' + tag) if tag else ''), cfg


def _cfgfile(key, q):
    import json
    with open('/tmp/gate-10069/sim/best.json') as f:
        d = json.load(f)
    return dict(d[key]['%.2f' % q])


def best(q):       # the best static depth x isolation of section A, per q
    return _cfgfile('best', q)


def winner(q):     # the best overall configuration of sections A-C, per q
    return _cfgfile('winner', q)


BASE = ('base', J(sus='J1d'))            # Jc-J1d as accepted: table depth, 2 x p95 deadline
REF = [('T', dict(base='T')), ('T+', dict(base='T+')), BASE]


def depth_pols(dl=NODL, isos=ISO, depths=DEPTHS):
    out = []
    for dn, dk in depths:
        for iname, ik in isos:
            kw = dict(ik); kw.update(dk); kw.update(dl)
            out.append(('d%s %s' % (dn, iname), J(**kw)))
    return out


def plan(name, arg=None):

        for q in QS:

            for env in ('off', 'on'):

            P.append(cell(q, env='off') + (depth_pols(isos=ISO[:2], depths=sub), 40))

        for q in QS:
            core = dict(sus='J1d') if name == 'B_table' else best(q)
            pols = [('B0', J(**core))]
            for x in (0.3, 0.5, 0.7):

            pols.append(('B3 hold', J(nostack=True, **core)))

                P.append(cell(q, tag='reverify=' + rv, reverify=rv) + (pols, 60))

    elif name == 'E':
        # the failing test names a request: right with accuracy a, nobody with probability 0.2
        sub = [d for d in DEPTHS if d[0] in ('+0', '+2', '+4', 'x2')]
        for q in QS:
            for a in (0.5, 0.8, 0.95):
                pols = [BASE]
                for dn, dk in sub:
                    kw = dict(sus='J1d'); kw.update(dk); kw.update(NODL)
                    pols.append(('d%s no naming' % dn, J(**kw)))
                    pols.append(('d%s confirm-first' % dn, J(naming='confirm', **kw)))
                    pols.append(('d%s block at once' % dn, J(naming='block', **kw)))
                P.append(cell(q, tag='a=%.2f' % a, name_acc=a) + (pols, 50))
    else:
        raise SystemExit('unknown plan ' + name)
    return P
