"""Tables from results/<plan>.json.  Usage: python report.py <plan> [...]  -> results/<plan>.txt"""
import os
import sys

sys.path.insert(0, '/tmp/gate-10069/sim')
import gridlib as G  # noqa: E402


def f(x, nd=1):
    return '-' if x is None else ('%.' + str(nd) + 'f') % x


def sizes(size):
    tot = sum(size) or 1
    return ' '.join('%d:%d%%' % (i, round(100.0 * size[i] / tot)) for i in range(26) if size[i] / tot >= 0.015)


def calib(c):
    out = []
    for b in range(10):
        n, sp, ps = c[b]
        if n >= 30:
            out.append('[%.1f-%.1f) n=%d pred %.2f obs %.2f' % (b / 10, (b + 1) / 10, n, sp / n, ps / n))
    return '; '.join(out)


def table(recs, cell, out):
    rows = [r for r in recs if r['cell'] == cell]
    if not rows:
        return
    base = rows[0]['pol']
    out.append('')
    out.append('### %s   (n=%d seeds per policy)' % (cell, rows[0]['n']))
    out.append('policy | landings/h | landings per verify-h | verifies/landing | d(land/verify-h) vs %s, paired | void%% (of which main-red%%) | idle%% | good wait h p50/p95 (still waiting%%) | bad: h to block p50/p95 (unresolved%%) | solo verifies/100 land | convictions/100 land right/WRONG | good blocked by own solo/100 land' % base)
    for r in rows:
        if r['pol'] == base:
            d = '—'
        else:
            dm, dc = G.paired(recs, cell, base, r['pol'], 'lvh_runs')
            d = '%+.3f ±%.3f' % (dm, dc)
        out.append('%s | %.2f ±%.2f | %.3f ±%.3f | %.3f ±%.3f | %s | %.0f (%.0f) | %.0f | %s/%s (%.0f) | %s/%s (%.0f) | %.0f | %.2f/%.2f | %.2f' % (
            r['pol'], r['thr'][0], r['thr'][1], r['lvh'][0], r['lvh'][1], r['vpl'][0], r['vpl'][1], d,
            100 * r['void_share'], 100 * r['mr_share'], 100 * r['idle_share'],
            f(r['good_p50']), f(r['good_p95']), 100 * r['good_cens'],
            f(r['bad_p50']), f(r['bad_p95']), 100 * r['bad_cens'],
            r['solo_per100'], r['conv_right_per100'], r['conv_wrong_per100'], r['false_block_per100']))
    out.append('')
    out.append('batch sizes (share of verifies started) and the membership rule (J clean batches: share where the rule left out a request the table would have taken / batch smaller than the table / larger than the table):')
    for r in rows:
        extra = ''
        if r['clean_per_run'] > 0:
            extra = ' | clean batches/run %.0f: rule excluded %.1f%%, below table %.1f%%, above table %.1f%%; largest group %d' % (
                r['clean_per_run'], 100 * r['rule_excl'], 100 * r['below_table'], 100 * r['above_table'], r['max_group'])
        out.append('  %s: %s%s' % (r['pol'], sizes(r['size']), extra))
    out.append('')
    out.append('calibration (predicted P(pass) of the tree at dispatch vs returned verdicts; bins with n>=30), HEAD = nothing unlanded ahead at dispatch, STACKED = stacked on unlanded entries:')
    for r in rows:
        for lab, key in (('HEAD', 'calib_head'), ('STACKED', 'calib_stack')):
            c = calib(r.get(key) or [[0, 0, 0]] * 10)
            if c:
                out.append('  %s %s: %s' % (r['pol'], lab, c))
    out.append('')
    out.append('verify time by fate (share of verify-hours): ' + ' | '.join(
        '%s: %s' % (r['pol'], ' '.join('%s %.0f' % (k, 100 * v) for k, v in r['cat'].items() if v >= 0.005)) for r in rows))
    out.append('suspect/clean entries per run: ' + ' | '.join(
        '%s: %s' % (r['pol'], ' '.join('%s %.0f' % kv for kv in sorted(r['kinds'].items()))) for r in rows))


def main(name):
    d = G.load(name)
    recs = d['recs']
    out = ['# plan %s (%.0f s)' % (name, d['elapsed'])]
    seen = []
    for r in recs:
        if r['cell'] not in seen:
            seen.append(r['cell'])
    for cell in seen:
        table(recs, cell, out)
    path = os.path.join(G.RES, name + '.txt')
    with open(path, 'w') as fh:
        fh.write('\n'.join(out) + '\n')
    print('wrote', path, len(out), 'lines')


if __name__ == '__main__':
    for nm in sys.argv[1:]:
        main(nm)
